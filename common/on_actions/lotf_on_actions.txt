on_actions = {
	on_startup = {
		effect = {
			#Refugees

			every_country = {
				limit = {
					exists = yes
					OR = {
						tag = ERR
						tag = ARA
						tag = CMP
						tag = MCH
						tag = WSC
					}
				}
				add_ideas = GENERIC_midwest_refugee_crisis
			}
			
		}
	}
	on_peaceconference_ended = {
		effect = {
			FROM = {
				if = {
					limit = {
						exists = yes
						NOT = {
							has_country_flag = keep_alive_after_peace
						}
					}
					ROOT = {
						annex_country = {
							target = PREV
							transfer_troops = no
						}
					}
				}
			}
		}
	}
	on_annex = {
		effect = {
			limit = {
				ROOT = IEF
				FROM = STL
            }
			complete_national_focus = IEF_proper_learning_experience
		}
	}
}
